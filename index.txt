    <script>
        // Gestione Stato Timer
        let phases = [];
        let currentPhaseIndex = 0;
        let timeRemainingSeconds = 0;
        let timerInterval = null;
        let isRunning = false;

        // Riferimenti DOM
        const displayTime = document.getElementById('time-left');
        const displayPhase = document.getElementById('current-phase-label');
        const btnStart = document.getElementById('btn-start');
        const btnPause = document.getElementById('btn-pause');
        const btnReset = document.getElementById('btn-reset');
        const soundToggle = document.getElementById('sound-toggle');
        const phasesList = document.getElementById('phases-list');
        const btnAddPhase = document.getElementById('btn-add-phase');

        // Audio Context per il bip sonoro (senza file esterni)
        function playBeep() {
            if (!soundToggle.checked) return;
            try {
                const AudioContext = window.AudioContext || window.webkitAudioContext;
                const ctx = new AudioContext();
                const oscillator = ctx.createOscillator();
                const gainNode = ctx.createGain();
                
                oscillator.type = 'sine';
                oscillator.frequency.setValueAtTime(880, ctx.currentTime); // 880Hz nota A5
                
                gainNode.gain.setValueAtTime(0.1, ctx.currentTime);
                gainNode.gain.exponentialRampToValueAtTime(0.00001, ctx.currentTime + 1);
                
                oscillator.connect(gainNode);
                gainNode.connect(ctx.destination);
                
                oscillator.start();
                oscillator.stop(ctx.currentTime + 1);
            } catch (e) {
                console.error("Errore Web Audio API:", e);
            }
        }

        // Formattazione del tempo in MM:SS
        function formatTime(totalSeconds) {
            const minutes = Math.floor(totalSeconds / 60);
            const seconds = totalSeconds % 60;
            return `${minutes.toString().padStart(2, '0')}:${seconds.toString().padStart(2, '0')}`;
        }

        function updateDisplay() {
            displayTime.textContent = formatTime(timeRemainingSeconds);
        }

        // Estrae i dati dagli input HTML per generare l'array dei cicli
        function parsePhasesFromInputs() {
            phases = [];
            const rows = document.querySelectorAll('.phase-row');
            rows.forEach(row => {
                const name = row.querySelector('.phase-name').value.trim() || 'Fase';
                const durationMinutes = parseInt(row.querySelector('.phase-duration').value) || 1;
                phases.push({ name, duration: durationMinutes * 60 });
            });
        }

        // Motore del timer
        function tick() {
            if (timeRemainingSeconds > 0) {
                timeRemainingSeconds--;
                updateDisplay();
            } else {
                playBeep();
                // Passa alla fase successiva (ciclo infinito)
                currentPhaseIndex = (currentPhaseIndex + 1) % phases.length;
                loadPhase(currentPhaseIndex);
            }
        }

        function loadPhase(index) {
            timeRemainingSeconds = phases[index].duration;
            displayPhase.textContent = phases[index].name;
            updateDisplay();
        }

        // Event Listeners dei controlli
        btnStart.addEventListener('click', () => {
            if (isRunning) return;
            
            // Carica le fasi solo se è l'inizio di un nuovo ciclo da fermo
            if (!timerInterval && timeRemainingSeconds === 0) {
                parsePhasesFromInputs();
                if (phases.length === 0) return;
                currentPhaseIndex = 0;
                loadPhase(currentPhaseIndex);
            }

            isRunning = true;
            btnStart.disabled = true;
            btnPause.disabled = false;
            
            // Disabilita modifiche durante l'esecuzione
            document.querySelectorAll('.phase-row input').forEach(inp => inp.disabled = true);
            
            timerInterval = setInterval(tick, 1000);
        });

        btnPause.addEventListener('click', () => {
            if (!isRunning) return;
            clearInterval(timerInterval);
            isRunning = false;
            btnStart.disabled = false;
            btnPause.disabled = true;
        });

        btnReset.addEventListener('click', () => {
            clearInterval(timerInterval);
            timerInterval = null;
            isRunning = false;
            timeRemainingSeconds = 0;
            
            btnStart.disabled = false;
            btnPause.disabled = true;
            
            displayPhase.textContent = "Pronto";
            displayTime.textContent = "00:00";
            
            // Riabilita modifiche
            document.querySelectorAll('.phase-row input').forEach(inp => inp.disabled = false);
        });

        // Aggiunta dinamica di nuove fasi
        btnAddPhase.addEventListener('click', () => {
            const row = document.createElement('div');
            row.className = 'phase-row';
            row.innerHTML = `
                <input type="text" value="Nuova Fase" class="phase-name" placeholder="Nome fase">
                <input type="number" value="5" class="phase-duration" min="1" placeholder="Min">
                <button onclick="this.parentElement.remove()" style="padding: 0.5rem; background: var(--danger-color);">X</button>
            `;
            phasesList.appendChild(row);
        });

        // Inizializzazione della vista al caricamento
        parsePhasesFromInputs();
        if(phases.length > 0) {
            displayTime.textContent = formatTime(phases[0].duration);
        }
    </script>
